# Global npm packages installed against the mise-managed Node.
# One package per line; blank lines and # comments are ignored.
#
# These replace the old nvm default-packages mechanism, which mise has no
# equivalent for. The language servers here are the ones Neovim enables in
# nvim/.config/nvim/lua/plugins/lsp.lua - the config deliberately does not use
# Mason, so this file is the single source of truth for what gets installed.

# Formatters / linters
@fsouza/prettierd
eslint_d
blade-formatter               # Laravel Blade templates

# Language servers
bash-language-server
emmet-ls
intelephense
vscode-langservers-extracted  # provides jsonls, html, cssls, eslint
@tailwindcss/language-server
@vue/language-server           # vue_ls, and bundles @vue/typescript-plugin
@vtsls/language-server         # vtsls

# makes vue_ls crash on startup with
#   TypeError: Cannot read properties of undefined (reading 'protocol')
typescript@5

# Required by nvim-treesitter (main branch) to compile parsers
tree-sitter-cli
